import {expect,it,vi,afterEach} from 'vitest';
import {clearFallenCharacterLifeCheckpoint,CHARACTER_LIFE_CHECKPOINT_SOCIETY_KEY as KEY} from '../character-life-session.js';
afterEach(()=>vi.unstubAllGlobals());
it('clears only the life slot after a confirmed fallen End Week',()=>{const values=new Map([[KEY,'saved'],['other','protected']]);const removeItem=vi.fn((key:string)=>values.delete(key));vi.stubGlobal('window',{localStorage:{getItem:(key:string)=>values.get(key)??null,removeItem}});for(const [success,stage] of [[false,'fallen'],[true,'none'],[true,'rebellion']] as const){expect(clearFallenCharacterLifeCheckpoint(success,stage)).toEqual({success:true});expect(values.get(KEY)).toBe('saved');}expect(removeItem).not.toHaveBeenCalled();expect(clearFallenCharacterLifeCheckpoint(true,'fallen')).toEqual({success:true});expect(removeItem.mock.calls).toEqual([[KEY]]);expect(values.get('other')).toBe('protected');expect(values.has(KEY)).toBe(false);});
it('contains storage failures',()=>{vi.stubGlobal('window',{get localStorage(){throw Error('private')}});expect(clearFallenCharacterLifeCheckpoint(true,'fallen')).toEqual({success:false,message:'Could not clear the local checkpoint. Browser storage is unavailable.'});});
